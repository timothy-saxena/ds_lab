# Question 1: Check whether a given integer is prime.
n <- as.integer(readline("Enter an integer: "))

if (is.na(n)) {
  stop("Please enter a whole number.")
} else if (n < 2) {
  cat(n, "is not prime.\n")
} else {
  divisors <- if (n < 4) integer(0) else 2:floor(sqrt(n))
  is_prime <- length(divisors) == 0 || !any(n %% divisors == 0)
  if (is_prime) cat(n, "is prime.\n") else cat(n, "is not prime.\n")
}

# Question 9: Calculator operations and common dataset functions.
calculator <- function() {
  
  repeat {
    
    num1 <- as.numeric(readline("Enter first number: "))
    num2 <- as.numeric(readline("Enter second number: "))
    
    cat("\n1. Addition")
    cat("\n2. Subtraction")
    cat("\n3. Multiplication")
    cat("\n4. Division")
    cat("\n5. Exit\n")
    
    choice <- as.integer(readline("Enter your choice: "))
    
    result <- switch(choice,
                     num1 + num2,
                     num1 - num2,
                     num1 * num2,
                     num1 / num2,
                     "exit")
    
    if (choice == 5) {
      cat("Calculator exited.\n")
      break
    }
    
    cat("Result =", result, "\n\n")
  }
}

       pch = 8,

# Read data
data <- read.csv("/Users/timothysaxena/Downloads/MGIT/Sem-V/8. DSL/Mid2/logistic regression/admissions.csv")

# Fit logistic regression
model <- glm(admit ~ gre + gpa + rank,
             data = data, family = binomial)

summary(model)

# Predict probabilities and classes
prob <- predict(model, type = "response")
predicted <- ifelse(prob >= 0.5, 1, 0)

# Confusion matrix and training accuracy
print(table(Actual = data$admit, Predicted = predicted))
print(mean(predicted == data$admit))

# Predict admission for a new student
new_student <- data.frame(
  gre = 700, gpa = 3.8, rank = factor(1)
)

p <- predict(model, newdata = new_student, type = "response")
print(p)
print(ifelse(p >= 0.5, 1, 0))

     xlab = "Student", ylab = "Admission probability",
     ylim = c(0, 1))
